\medskip\noindent\textbf{Part 2 (13 points): Binary Dynamics and Error Propagation}

\begin{description}
\item[(a)] Kepler's 3rd law:
  \[
  P_b^2 = \frac{4\pi^2}{G\left(M_{\mathrm{PSR}} + M_{\mathrm{WD}}\right)}
    \left(a_p + a_{\mathrm{WD}}\right)^3
  \]
  \hfill(2 points)

  Center of mass momentum conservation:
  \[
  a_{\mathrm{WD}} = a_p\frac{M_{\mathrm{PSR}}}{M_{\mathrm{WD}}}
  \]
  \hfill(2 points)

  Substitute $a_{\mathrm{WD}}$ into the above equation and isolate variables
  to derive the relation:
  \[
  f_M = \frac{M_{\mathrm{WD}}^3}{\left(M_{\mathrm{PSR}} + M_{\mathrm{WD}}\right)^2}
      = \frac{4\pi^2 a_p^3}{G P_b^2} \quad (1)
  \]
  \hfill(1 point)

  Computation:
  \[
  f_M \approx 5.927 \times 10^{27}\,\mathrm{kg} \approx 0.002975\,\mathrm{M}_\odot
  \]
  \hfill(1 point)

\item[(b)] Rearrange (1) to solve for $M_{\mathrm{PSR}}$:
  \[
  M_{\mathrm{PSR}} = \sqrt{\frac{M_{\mathrm{WD}}^3}{f_M}} - M_{\mathrm{WD}}
  \quad (2)
  \]
  \hfill(1 point)

  Substitute values to obtain:
  \[
  M_{\mathrm{PSR}} \approx 2.20\,\mathrm{M}_\odot
  \]
  \hfill(1 point)

  Differentiate both sides of (2) with respect to $M_{\mathrm{WD}}$:
  \[
  \frac{\Delta M_{\mathrm{PSR}}}{\Delta M_{\mathrm{WD}}}
    = \frac{3}{2}\sqrt{\frac{M_{\mathrm{WD}}}{f_M}} - 1 \approx 13.08
  \]
  \hfill(2 points)

  Error computation:
  \[
  \Delta M_{\mathrm{PSR}} \approx 13.08 \times \Delta M_{\mathrm{WD}}
    \approx 0.27\,\mathrm{M}_\odot
  \]
  \hfill(1 point)

  Conclusion:
  \[
  M_{\mathrm{PSR}} = 2.20 \pm 0.27\,\mathrm{M}_\odot
  \]

\item[(c)] Compute:
  \[
  n_\sigma = \frac{M_{\mathrm{PSR}} - M_{\mathrm{limit}}}{\Delta M_{\mathrm{PSR}}}
    = \frac{2.20 - 2.0}{0.27} = 0.74
  \]
  \hfill(1 point)

  Answer: No \hfill(1 point)

  \emph{Since $n_\sigma < 3$, the measurement does not provide sufficient
  statistical significance to rule out the soft EoS theory.}
\end{description}
