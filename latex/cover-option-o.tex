    % ---------------- P1: flat rotation curves (top left) ----------------
    \begin{scope}[shift={(2.6,12.6)}]
      \draw[glav,opacity=0.35,line width=0.5pt,
            dash pattern=on 1.6pt off 2.2pt,-stealth] (-0.15,0) -- (6.4,0);
      \draw[glav,opacity=0.35,line width=0.5pt,
            dash pattern=on 1.6pt off 2.2pt,-stealth] (0,-0.15) -- (0,4.1);
      \draw[glav,opacity=0.6,line width=0.7pt,smooth,
            dash pattern=on 2.4pt off 2.6pt]
        plot[domain=1.5:6.1,samples=60] (\x,{3.135/sqrt(\x)});
      \draw[gpink,line width=1.2pt,smooth]
        plot[domain=0.02:6.1,samples=90]
        (\x,{2.9*\x/sqrt(\x*\x+0.64) + 0.05*sin(\x*150)});
      \fill[gwhite] (5.0,{2.9*5.0/sqrt(25.0+0.64)+0.05*sin(5.0*150)})
        circle (2.2pt);
      \draw[gmagenta,line width=0.7pt,stealth-stealth]
        (5.6,{3.135/sqrt(5.6)+0.14})
          -- (5.6,{2.9*5.6/sqrt(5.6*5.6+0.64)+0.05*sin(5.6*150)-0.14});
      \panelcaption{flat rotation curves}
    \end{scope}

    % ---------------- P2: gravitational lensing (top right) ----------------
    \begin{scope}[shift={(11.4,12.6)}]
      \draw[glav,opacity=0.35,line width=0.5pt,

        (0.3,2.1) .. controls (2.2,3.3) and (4.4,3.3) .. (6.1,2.13);
      \draw[gpink,line width=1.2pt,-stealth]
        (0.3,2.1) .. controls (2.2,0.9) and (4.4,0.9) .. (6.1,2.07);

      \fill[gmagenta,opacity=0.4] (0.3,2.1) circle (0.26);
      \fill[gpink] (0.3,2.1) circle (0.15);
      \fill[gwhite] (6.2,2.1) circle (2.2pt);
      \panelcaption{gravitational lensing}
    \end{scope}

    % ---------------- P3: colliding clusters (bottom left) ----------------
    \begin{scope}[shift={(2.6,5.4)}]
      % Lagging gas, still in the middle.
      \fill[gpink,opacity=0.28] (2.55,2.1) ellipse (0.62 and 0.5);

      \draw[glav,opacity=0.7,line width=0.6pt,-stealth]
        (5.35,3.15) -- (6.05,3.15);
      \panelcaption{colliding clusters}
    \end{scope}

      \draw[gpink,line width=0.4cm]
        ($(3.25,2.35)+(-252:1.3)$) arc (-252:-270:1.3);
      \node[gmagenta,font=\normalsize\sffamily\bfseries] at (3.25,2.55)

      \node[glav,font=\footnotesize\sffamily] at (3.25,1.95) {dark matter};
      \node[glav,opacity=0.8,font=\footnotesize\sffamily,anchor=west]
        at (4.95,1.28) {dark energy};
      \node[glav,opacity=0.8,font=\footnotesize\sffamily] at (2.5,4.15)

  \end{scope}
