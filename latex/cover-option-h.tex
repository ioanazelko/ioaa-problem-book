% \moonphase{k}{radius} : phase k = 0..7 (0 new, 4 full), lit side per k.
\newcommand{\moonphase}[2]{%
  \fill[hplum] (0,0) circle (#2);
  \ifnum#1>0
    \ifnum#1<4
      % Waxing: lit on the right; terminator x-radius -r*cos(45k).
      \pgfmathsetmacro{\xr}{-#2*cos(#1*45)}%
      \fill[hlit] (0,-#2) arc (-90:90:#2)
        arc (90:270:{\xr} and {#2}) -- cycle;
    \else
      \ifnum#1=4
        \fill[hlit] (0,0) circle (#2);
      \else
        % Waning: lit on the left.
        \pgfmathsetmacro{\xr}{#2*cos((#1-4)*45)}%
        \fill[hlit] (0,#2) arc (90:270:#2)
          arc (270:450:{\xr} and {#2}) -- cycle;
      \fi
    \fi
  \fi
  \draw[hplum,line width=0.8pt] (0,0) circle (#2);}
