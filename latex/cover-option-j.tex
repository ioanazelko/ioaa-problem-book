% Cover option J: "cosmic expansion" -- same dusk style as option G, but the
% artwork is the expansion history of the universe: a flaring cone from the
% Big Bang to today, epochs marked along the boundary.
\documentclass[11pt,a4paper]{article}
\usepackage[margin=0cm]{geometry}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{tikz}
\usetikzlibrary{calc}
\pagestyle{empty}

\definecolor{gtop}{HTML}{170D2B}
\definecolor{gbottom}{HTML}{51285F}
\definecolor{gpink}{HTML}{F48FC0}
\definecolor{gmagenta}{HTML}{E0559E}
\definecolor{glav}{HTML}{CBB4DD}
\definecolor{gwhite}{HTML}{F8F3FB}

% Cone half-width: sqrt growth (decelerating) + cubic term (dark energy).
\newcommand{\conew}[1]{(0.5*sqrt(#1)+0.0008*(#1)^3)}

\begin{document}
\begin{tikzpicture}[remember picture,overlay]
  \shade[top color=gtop,bottom color=gbottom]
    (current page.south west) rectangle (current page.north east);

  \begin{scope}[shift={(current page.south west)}]
    \begin{scope}[shift={(13.6,3.0)}]
      % Time axis.
      \draw[glav,opacity=0.35,line width=0.5pt,
            dash pattern=on 1.6pt off 2.2pt,-stealth] (0,-0.7) -- (0,17.0);
      \node[glav,opacity=0.7,font=\footnotesize\sffamily,anchor=west]
        at (0.3,16.8) {time};
      % Boundary of the observable universe.
      \draw[gpink,line width=1.3pt,smooth]
        plot[domain=0.02:15.8,samples=160] ({\conew{\x}},\x);
      \draw[gpink,line width=1.3pt,smooth]
        plot[domain=0.02:15.8,samples=160] ({-\conew{\x}},\x);
      % Faint cross-sections, and the rim "today".
      \foreach \yy in {2.2,5.5,8.5,12.0}
        \draw[glav,opacity=0.30,line width=0.5pt]
          (0,\yy) ellipse ({\conew{\yy}} and {0.14*\conew{\yy}});
      \draw[gpink,line width=1.0pt,opacity=0.9]
        (0,15.8) ellipse ({\conew{15.8}} and {0.14*\conew{15.8}});
      % Galaxies inside the late universe.
      \foreach \gx/\gy/\gs/\gc in {
        -1.2/9.5/1.1/gwhite, 0.8/10.7/0.9/glav, 1.9/12.4/1.2/gwhite,
        -2.4/13.1/0.9/glav, 0.2/14.2/1.1/gwhite, 2.9/14.9/0.9/glav,
        -3.3/15.0/1.2/gwhite, -0.9/12.0/0.9/gwhite, 3.4/13.6/0.8/glav}
        \fill[\gc] (\gx,\gy) circle (\gs pt);
      % The Big Bang, glowing.
      \fill[gmagenta,opacity=0.22] (0,0) circle (0.62);
      \fill[gmagenta,opacity=0.4] (0,0) circle (0.4);
      \fill[gpink] (0,0) circle (0.22);
      \node[gpink,font=\footnotesize\sffamily,anchor=west] at (0.85,-0.4)
        {Big Bang};
      % Epochs, dotted along the boundary.
      \fill[gwhite] ({\conew{0.55}},0.55) circle (2.6pt);
      \node[glav,font=\footnotesize\sffamily,anchor=west]
        at ({\conew{0.55}+0.35},0.55) {inflation};
      \fill[gwhite] ({-\conew{2.2}},2.2) circle (2.6pt);
      \node[glav,font=\footnotesize\sffamily,anchor=east]
        at ({-\conew{2.2}-0.35},2.2) {CMB released};
      \fill[gwhite] ({\conew{5.5}},5.5) circle (2.6pt);
      \node[glav,font=\footnotesize\sffamily,anchor=west]
        at ({\conew{5.5}+0.35},5.5) {first stars};
      \fill[gwhite] ({-\conew{8.5}},8.5) circle (2.6pt);
      \node[glav,font=\footnotesize\sffamily,anchor=east]
        at ({-\conew{8.5}-0.35},8.5) {galaxies assemble};
      \fill[gwhite] ({-\conew{12.0}},12.0) circle (2.6pt);
      \node[glav,font=\footnotesize\sffamily,anchor=east]
        at ({-\conew{12.0}-0.35},12.0) {expansion accelerates};
      \fill[gwhite] ({-\conew{15.8}},15.8) circle (2.6pt);
      \node[glav,font=\footnotesize\sffamily,anchor=east]
        at ({-\conew{15.8}-0.35},15.8) {today};
    \end{scope}
  \end{scope}

  % ---------------- title block, left ----------------
