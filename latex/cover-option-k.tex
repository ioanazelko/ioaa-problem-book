% Planck-shaped curve normalized to peak height h at x = xp
% (Wien-form exponent keeps every intermediate value small for pgf math).
% \planck{h}{xp}{opacity}{line width}
\newcommand{\planck}[4]{%
  \draw[gpink,opacity=#3,line width=#4,smooth]
    plot[domain=0.85:15.8,samples=170]
    (\x,{#1*exp(-4.965*#2/\x + 4.965 + 5*ln(#2/\x))});}
